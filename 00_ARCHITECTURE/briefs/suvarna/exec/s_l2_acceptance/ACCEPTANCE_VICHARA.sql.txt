-- ACCEPTANCE_VICHARA.sql.txt (lane L-KARANAJALA, N-143 option B; draft for the S-L2 rehearsal runbook)
--
-- Acceptance criterion (SS B3 ruling 2(c)): after S-L2, 100% of the vichara citations in bodha_cgm_edges, bodha_mechanisms and the
-- RM rows (bodha_rm_dasha_windowed_prescriptions, schedule_jsonb.leverage_index) of the canonical chart RESOLVE to EXACTLY ONE
-- chart_vichara row through the read-only resolver (migration 1295: view public.vw_chart_vichara_token, one row per chart_vichara
-- row, joined on chart_id + vichara_token), and every cited value is token-shaped (16 lowercase hex), none serial-shaped.
--
-- PASS condition after S-L2 (canonical chart 482012f1-710e-4a25-994a-93821f5871aa):
--   V1, V2, V3: citations = resolved_to_exactly_one_row, unresolved = 0, resolved_to_several_rows = 0
--   V4: surplus_rows = 0 (the canonical chart has no two vichara rows with one token)
--   V5: returns no rows
--   V6: citations = token_shaped, serial_shaped = 0
-- For a chart whose chart_vichara still holds exact duplicate rows (the two never-rebuilt charts: 666 / 671 identical pairs), V1-V3
-- would report resolved_to_several_rows > 0: those pairs are content-identical (every column but id), so for such charts the
-- statement to assert is 'exactly one DISTINCT natural key per token' (count(DISTINCT natural key) = 1); the canonical chart is
-- the acceptance subject.
-- Each statement is ONE read-only SELECT with no semicolon inside (rq.sh shape). The RM extraction accepts either key
-- (vichara_token after lane B4, vichara_row_id before) so the same statement measures before and after.
--
-- SECTION A: the FINAL form (run after migration 1295 is applied; reads public.vw_chart_vichara_token).

-- V1_bodha_cgm_edges
SELECT 'bodha_cgm_edges' AS store, count(*) AS citations, count(*) FILTER (WHERE r.n = 1) AS resolved_to_exactly_one_row, count(*) FILTER (WHERE r.n IS NULL) AS unresolved, count(*) FILTER (WHERE r.n > 1) AS resolved_to_several_rows FROM (SELECT e.chart_id, c AS cite FROM bodha_cgm_edges e CROSS JOIN LATERAL unnest(e.constituent_ga_vichara_ids_array) AS c WHERE e.chart_id = '482012f1-710e-4a25-994a-93821f5871aa') x LEFT JOIN (SELECT chart_id, vichara_token, count(*) AS n FROM vw_chart_vichara_token r0 GROUP BY 1, 2) r ON r.chart_id = x.chart_id AND r.vichara_token = x.cite

-- V2_bodha_mechanisms
SELECT 'bodha_mechanisms' AS store, count(*) AS citations, count(*) FILTER (WHERE r.n = 1) AS resolved_to_exactly_one_row, count(*) FILTER (WHERE r.n IS NULL) AS unresolved, count(*) FILTER (WHERE r.n > 1) AS resolved_to_several_rows FROM (SELECT m.chart_id, c AS cite FROM bodha_mechanisms m CROSS JOIN LATERAL unnest(m.constituent_ga_vichara_ids_array) AS c WHERE m.chart_id = '482012f1-710e-4a25-994a-93821f5871aa') x LEFT JOIN (SELECT chart_id, vichara_token, count(*) AS n FROM vw_chart_vichara_token r0 GROUP BY 1, 2) r ON r.chart_id = x.chart_id AND r.vichara_token = x.cite

-- V3_bodha_rm_dasha_windowed_prescriptions
SELECT 'bodha_rm_dasha_windowed_prescriptions' AS store, count(*) AS citations, count(*) FILTER (WHERE r.n = 1) AS resolved_to_exactly_one_row, count(*) FILTER (WHERE r.n IS NULL) AS unresolved, count(*) FILTER (WHERE r.n > 1) AS resolved_to_several_rows FROM (SELECT p.chart_id, coalesce(p.schedule_jsonb->'leverage_index'->>'vichara_token', p.schedule_jsonb->'leverage_index'->>'vichara_row_id') AS cite FROM bodha_rm_dasha_windowed_prescriptions p WHERE p.chart_id = '482012f1-710e-4a25-994a-93821f5871aa' AND p.schedule_jsonb->'leverage_index' IS NOT NULL) x LEFT JOIN (SELECT chart_id, vichara_token, count(*) AS n FROM vw_chart_vichara_token r0 GROUP BY 1, 2) r ON r.chart_id = x.chart_id AND r.vichara_token = x.cite

-- V4_resolver_unique_on_canonical_chart
SELECT count(*) AS rows_in_chart, count(DISTINCT vichara_token) AS distinct_tokens, count(*) - count(DISTINCT vichara_token) AS surplus_rows FROM vw_chart_vichara_token r WHERE r.chart_id = '482012f1-710e-4a25-994a-93821f5871aa'

-- V5_unresolved_citation_sample_edges
SELECT x.cite, count(*) AS uses FROM (SELECT e.chart_id, c AS cite FROM bodha_cgm_edges e CROSS JOIN LATERAL unnest(e.constituent_ga_vichara_ids_array) AS c WHERE e.chart_id = '482012f1-710e-4a25-994a-93821f5871aa') x LEFT JOIN (SELECT chart_id, vichara_token, count(*) AS n FROM vw_chart_vichara_token r0 GROUP BY 1, 2) r ON r.chart_id = x.chart_id AND r.vichara_token = x.cite WHERE r.n IS NULL GROUP BY 1 ORDER BY 2 DESC, 1 LIMIT 10

-- V6_every_citation_is_token_shaped
SELECT count(*) AS citations, count(*) FILTER (WHERE x.cite ~ '^[0-9a-f]{16}$') AS token_shaped, count(*) FILTER (WHERE x.cite ~ '^[0-9]+$') AS serial_shaped FROM (SELECT c AS cite FROM bodha_cgm_edges e CROSS JOIN LATERAL unnest(e.constituent_ga_vichara_ids_array) AS c WHERE e.chart_id = '482012f1-710e-4a25-994a-93821f5871aa' UNION ALL SELECT c FROM bodha_mechanisms m CROSS JOIN LATERAL unnest(m.constituent_ga_vichara_ids_array) AS c WHERE m.chart_id = '482012f1-710e-4a25-994a-93821f5871aa') x
