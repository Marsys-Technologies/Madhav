-- ACCEPTANCE_VICHARA.sql.txt (lane L-KARANAJALA, N-143 option B; for the pinned S-L2 acceptance set)
--
-- No database object resolves a token (SS ruling AVOID DDL): the single definition of the token is the committed expression
-- 00_ARCHITECTURE/briefs/suvarna/exec/s_l2_acceptance/vichara_token_expression.sql, INLINED below in every statement as a derived table over chart_vichara; a test requires the exact text.
-- Criterion: after S-L2, every vichara citation of the canonical chart (482012f1-710e-4a25-994a-93821f5871aa) in bodha_cgm_edges, bodha_mechanisms and the RM rows
-- (bodha_rm_dasha_windowed_prescriptions.schedule_jsonb.leverage_index.vichara_token) resolves to EXACTLY ONE chart_vichara row, every cited
-- value is token-shaped (16 lowercase hex), none serial-shaped, and no RM row carries a random id.
-- PASS: V1, V2, V3b: citations = resolved_to_exactly_one_row, unresolved = 0, resolved_to_several_rows = 0. V3a: with_vichara_serial = 0,
-- with_dasha_row_id = 0, with_uuid_v4_token = 0. V3c: with_structured_dasha_key = rm_rows (the dasha is cited by natural key
-- system_id, ayanamsha_id, level_n, start_iso, kp_sublevel). NOTE: bo_upaya no longer writes the dasha-windowed rows (DP-SD-015: L3-only), so
-- after S-L2 rm_rows is expected to be 0 and V3a/V3b/V3c pass trivially: read rm_rows first; the legacy-id check V3a is what is non-vacuous today.
-- V4: surplus_rows = 0. V5: returns no rows. V6: citations = token_shaped, serial_shaped = 0.
-- Older never-rebuilt charts hold exact duplicate rows (666 / 671 identical pairs): there assert exactly one DISTINCT natural key per token.
-- Each statement is ONE read-only SELECT, no semicolon inside.

-- V1_bodha_cgm_edges
SELECT 'bodha_cgm_edges' AS store, count(*) AS citations, count(*) FILTER (WHERE r.n = 1) AS resolved_to_exactly_one_row, count(*) FILTER (WHERE r.n IS NULL) AS unresolved, count(*) FILTER (WHERE r.n > 1) AS resolved_to_several_rows FROM (SELECT e.chart_id, c AS cite FROM bodha_cgm_edges e CROSS JOIN LATERAL unnest(e.constituent_ga_vichara_ids_array) AS c WHERE e.chart_id = '482012f1-710e-4a25-994a-93821f5871aa') x LEFT JOIN (SELECT chart_id, vichara_token, count(*) AS n FROM (SELECT v.chart_id, v.id, left(encode(sha256(convert_to( '[' || concat_ws(',', coalesce(to_json(v.ayanamsha_id)::text, 'null'), coalesce(to_json(v.vichara_family)::text, 'null'), coalesce(to_json(v.subject)::text, 'null'), coalesce(to_json(v.actor)::text, 'null'), coalesce(to_json(v.target)::text, 'null'), coalesce(to_json(v.domain)::text, 'null'), coalesce(to_json(v.varga_id)::text, 'null'), coalesce(to_json(v.varga)::text, 'null'), coalesce(to_json(v.value_text)::text, 'null'), CASE WHEN v.value_num IS NULL THEN 'null' WHEN v.value_num = 'NaN'::numeric THEN '"NaN"' ELSE trim_scale(v.value_num)::text END, coalesce(v.value_jsonb::text, 'null'), coalesce(to_json(v.constituent_facts_array)::text, 'null') ) || ']', 'UTF8')), 'hex'), 16) AS vichara_token FROM chart_vichara v) r0 GROUP BY 1, 2) r ON r.chart_id = x.chart_id AND r.vichara_token = x.cite

-- V2_bodha_mechanisms
SELECT 'bodha_mechanisms' AS store, count(*) AS citations, count(*) FILTER (WHERE r.n = 1) AS resolved_to_exactly_one_row, count(*) FILTER (WHERE r.n IS NULL) AS unresolved, count(*) FILTER (WHERE r.n > 1) AS resolved_to_several_rows FROM (SELECT m.chart_id, c AS cite FROM bodha_mechanisms m CROSS JOIN LATERAL unnest(m.constituent_ga_vichara_ids_array) AS c WHERE m.chart_id = '482012f1-710e-4a25-994a-93821f5871aa') x LEFT JOIN (SELECT chart_id, vichara_token, count(*) AS n FROM (SELECT v.chart_id, v.id, left(encode(sha256(convert_to( '[' || concat_ws(',', coalesce(to_json(v.ayanamsha_id)::text, 'null'), coalesce(to_json(v.vichara_family)::text, 'null'), coalesce(to_json(v.subject)::text, 'null'), coalesce(to_json(v.actor)::text, 'null'), coalesce(to_json(v.target)::text, 'null'), coalesce(to_json(v.domain)::text, 'null'), coalesce(to_json(v.varga_id)::text, 'null'), coalesce(to_json(v.varga)::text, 'null'), coalesce(to_json(v.value_text)::text, 'null'), CASE WHEN v.value_num IS NULL THEN 'null' WHEN v.value_num = 'NaN'::numeric THEN '"NaN"' ELSE trim_scale(v.value_num)::text END, coalesce(v.value_jsonb::text, 'null'), coalesce(to_json(v.constituent_facts_array)::text, 'null') ) || ']', 'UTF8')), 'hex'), 16) AS vichara_token FROM chart_vichara v) r0 GROUP BY 1, 2) r ON r.chart_id = x.chart_id AND r.vichara_token = x.cite

-- V3a_rm_rows_carry_no_random_id
SELECT count(*) AS rm_rows, count(*) FILTER (WHERE p.schedule_jsonb->'leverage_index' ? 'vichara_row_id') AS with_vichara_serial, count(*) FILTER (WHERE p.schedule_jsonb->'dasha_runway' ? 'dasha_row_id') AS with_dasha_row_id, count(*) FILTER (WHERE p.schedule_jsonb::text ~ '[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}') AS with_uuid_v4_token FROM bodha_rm_dasha_windowed_prescriptions p WHERE p.chart_id = '482012f1-710e-4a25-994a-93821f5871aa'

-- V3b_rm_vichara_tokens_resolve
SELECT 'bodha_rm_dasha_windowed_prescriptions' AS store, count(*) AS citations, count(*) FILTER (WHERE r.n = 1) AS resolved_to_exactly_one_row, count(*) FILTER (WHERE r.n IS NULL) AS unresolved, count(*) FILTER (WHERE r.n > 1) AS resolved_to_several_rows FROM (SELECT p.chart_id, p.schedule_jsonb->'leverage_index'->>'vichara_token' AS cite FROM bodha_rm_dasha_windowed_prescriptions p WHERE p.chart_id = '482012f1-710e-4a25-994a-93821f5871aa' AND p.schedule_jsonb->'leverage_index'->>'vichara_token' IS NOT NULL) x LEFT JOIN (SELECT chart_id, vichara_token, count(*) AS n FROM (SELECT v.chart_id, v.id, left(encode(sha256(convert_to( '[' || concat_ws(',', coalesce(to_json(v.ayanamsha_id)::text, 'null'), coalesce(to_json(v.vichara_family)::text, 'null'), coalesce(to_json(v.subject)::text, 'null'), coalesce(to_json(v.actor)::text, 'null'), coalesce(to_json(v.target)::text, 'null'), coalesce(to_json(v.domain)::text, 'null'), coalesce(to_json(v.varga_id)::text, 'null'), coalesce(to_json(v.varga)::text, 'null'), coalesce(to_json(v.value_text)::text, 'null'), CASE WHEN v.value_num IS NULL THEN 'null' WHEN v.value_num = 'NaN'::numeric THEN '"NaN"' ELSE trim_scale(v.value_num)::text END, coalesce(v.value_jsonb::text, 'null'), coalesce(to_json(v.constituent_facts_array)::text, 'null') ) || ']', 'UTF8')), 'hex'), 16) AS vichara_token FROM chart_vichara v) r0 GROUP BY 1, 2) r ON r.chart_id = x.chart_id AND r.vichara_token = x.cite

-- V3c_rm_dasha_natural_key_shape
SELECT count(*) AS rm_rows, count(*) FILTER (WHERE p.schedule_jsonb->'dasha_runway'->'dasha_key' ?& ARRAY['system_id','ayanamsha_id','level_n','start_iso','kp_sublevel']) AS with_structured_dasha_key FROM bodha_rm_dasha_windowed_prescriptions p WHERE p.chart_id = '482012f1-710e-4a25-994a-93821f5871aa'

-- V4_token_unique_on_canonical_chart
SELECT count(*) AS rows_in_chart, count(DISTINCT vichara_token) AS distinct_tokens, count(*) - count(DISTINCT vichara_token) AS surplus_rows FROM (SELECT v.chart_id, left(encode(sha256(convert_to( '[' || concat_ws(',', coalesce(to_json(v.ayanamsha_id)::text, 'null'), coalesce(to_json(v.vichara_family)::text, 'null'), coalesce(to_json(v.subject)::text, 'null'), coalesce(to_json(v.actor)::text, 'null'), coalesce(to_json(v.target)::text, 'null'), coalesce(to_json(v.domain)::text, 'null'), coalesce(to_json(v.varga_id)::text, 'null'), coalesce(to_json(v.varga)::text, 'null'), coalesce(to_json(v.value_text)::text, 'null'), CASE WHEN v.value_num IS NULL THEN 'null' WHEN v.value_num = 'NaN'::numeric THEN '"NaN"' ELSE trim_scale(v.value_num)::text END, coalesce(v.value_jsonb::text, 'null'), coalesce(to_json(v.constituent_facts_array)::text, 'null') ) || ']', 'UTF8')), 'hex'), 16) AS vichara_token FROM chart_vichara v WHERE v.chart_id = '482012f1-710e-4a25-994a-93821f5871aa') r

-- V5_unresolved_edge_citation_sample
SELECT x.cite, count(*) AS uses FROM (SELECT e.chart_id, c AS cite FROM bodha_cgm_edges e CROSS JOIN LATERAL unnest(e.constituent_ga_vichara_ids_array) AS c WHERE e.chart_id = '482012f1-710e-4a25-994a-93821f5871aa') x LEFT JOIN (SELECT chart_id, vichara_token, count(*) AS n FROM (SELECT v.chart_id, v.id, left(encode(sha256(convert_to( '[' || concat_ws(',', coalesce(to_json(v.ayanamsha_id)::text, 'null'), coalesce(to_json(v.vichara_family)::text, 'null'), coalesce(to_json(v.subject)::text, 'null'), coalesce(to_json(v.actor)::text, 'null'), coalesce(to_json(v.target)::text, 'null'), coalesce(to_json(v.domain)::text, 'null'), coalesce(to_json(v.varga_id)::text, 'null'), coalesce(to_json(v.varga)::text, 'null'), coalesce(to_json(v.value_text)::text, 'null'), CASE WHEN v.value_num IS NULL THEN 'null' WHEN v.value_num = 'NaN'::numeric THEN '"NaN"' ELSE trim_scale(v.value_num)::text END, coalesce(v.value_jsonb::text, 'null'), coalesce(to_json(v.constituent_facts_array)::text, 'null') ) || ']', 'UTF8')), 'hex'), 16) AS vichara_token FROM chart_vichara v) r0 GROUP BY 1, 2) r ON r.chart_id = x.chart_id AND r.vichara_token = x.cite WHERE r.n IS NULL GROUP BY 1 ORDER BY 2 DESC, 1 LIMIT 10

-- V6_every_citation_is_token_shaped
SELECT count(*) AS citations, count(*) FILTER (WHERE x.cite ~ '^[0-9a-f]{16}$') AS token_shaped, count(*) FILTER (WHERE x.cite ~ '^[0-9]+$') AS serial_shaped FROM (SELECT c AS cite FROM bodha_cgm_edges e CROSS JOIN LATERAL unnest(e.constituent_ga_vichara_ids_array) AS c WHERE e.chart_id = '482012f1-710e-4a25-994a-93821f5871aa' UNION ALL SELECT c FROM bodha_mechanisms m CROSS JOIN LATERAL unnest(m.constituent_ga_vichara_ids_array) AS c WHERE m.chart_id = '482012f1-710e-4a25-994a-93821f5871aa') x
