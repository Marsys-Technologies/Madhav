-- S1_ga_structural_integrity_text_is_the_1326_text
SELECT md5(integrity_check_sql) AS md5, length(integrity_check_sql) AS chars, position('WHEN re.retrograde_flag = ''retrograde'' THEN ''weak''' IN integrity_check_sql) AS old_b4_line_position, position('a.fact_subject NOT IN (''RAH_MEAN'', ''KET_MEAN'') THEN ''weak''' IN integrity_check_sql) AS node_exclusion_position FROM asset_registry WHERE asset_id = 'ga_structural'

-- S2_every_canonical_node_composite_row_is_neutral_whatever_the_retrograde_flag_says
SELECT count(*) AS node_rows, count(*) FILTER (WHERE c.fact_value_text <> 'neutral') AS non_neutral_node_rows, count(*) FILTER (WHERE p.fact_value_text = 'retrograde') AS nodes_flagged_retrograde FROM chart_facts c JOIN chart_facts p ON p.chart_id = c.chart_id AND p.ayanamsha_id = c.ayanamsha_id AND p.fact_category = 'graha_position' AND p.fact_subject = c.fact_subject AND p.fact_key = 'retrograde_flag' WHERE c.chart_id = '482012f1-710e-4a25-994a-93821f5871aa' AND c.fact_category = 'graha_composite_state_classification' AND c.fact_subject IN ('RAH_MEAN', 'KET_MEAN')

-- S3_no_retrograde_tara_graha_is_classified_neutral_the_downgrade_still_holds_for_them
SELECT count(*) AS retrograde_tara_rows, count(*) FILTER (WHERE c.fact_value_text = 'neutral') AS retrograde_taras_left_neutral FROM chart_facts c JOIN chart_facts p ON p.chart_id = c.chart_id AND p.ayanamsha_id = c.ayanamsha_id AND p.fact_category = 'graha_position' AND p.fact_subject = c.fact_subject AND p.fact_key = 'retrograde_flag' AND p.fact_value_text = 'retrograde' WHERE c.chart_id = '482012f1-710e-4a25-994a-93821f5871aa' AND c.fact_category = 'graha_composite_state_classification' AND c.fact_subject IN ('SUN', 'MOON', 'MAR', 'MER', 'JUP', 'VEN', 'SAT')
