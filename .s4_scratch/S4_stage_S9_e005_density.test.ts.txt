    const sigTotal = totalSignalCitations(READING)
    console.log(`[E-005] countSignalCitations (SIG.MSR.NNN pattern) = ${sigCount}, totalSignalCitations = ${sigTotal}`)
    // The footnote DEFINITIONS in this real reading are raw evidence-row UUIDs,
    // not SIG.MSR.NNN ids -- so the production signal-id counter finds ZERO,
    // even though 7 real footnotes are structurally present.
    expect(sigCount).toBe(0)

    console.log(`[E-005] extractCitations() found: ${JSON.stringify(extracted)}`)

    // WARN/ERROR -- i.e. the gate would say nothing is wrong even though the

    expect(result.gate_result).toBe('PASS')
  })

    // dignities, yogas, dasha dates, structural claims) -- counted manually,
